Table 表格 ​

表格组件，基于 Element Plus Table 增强分页、选择、操作列、筛选、拖拽、虚拟滚动和空状态。

导出 ​

DasTable、DasTableRef、DasTableColumn、DasTableColumnRef

基础用法 ​

基础用法
展示表格选择、加载、高亮行、分页总数和操作列事件。
<template>
  <div class="table-demo">
    <div class="table-demo__toolbar">
      <DasButton btn-type="default" size="small" @click="loading = !loading">{{ loading ? '关闭' : '开启' }}加载</DasButton>
      <DasButton btn-type="default" size="small" @click="highlightId = highlightId === 2 ? 3 : 2">切换高亮行</DasButton>
      <span>已选 {{ selectionRows.length }} 项</span>
    </div>
    <DasTable
      v-model:selection-rows="selectionRows"
      v-model:loading="loading"
      :data="data"
      :total="data.length"
      :highlight-row-id="[highlightId]"
      row-key="id"
      border
      outer-border
      :selection="{ value: 'id', label: 'name', showSelectCount: true }"
      :operation="operation"
      :operation-options="{ width: { zh: 160 }, minWidth: { zh: 160 } }"
      @operation-click="handleOperation"
    >
      <DasTableColumn type="index" label="序号" width="70" />
      <DasTableColumn prop="name" label="名称" min-width="150" />
      <DasTableColumn prop="status" label="状态" min-width="120" />
      <DasTableColumn prop="owner" label="负责人" min-width="120" />
    </DasTable>
    <div class="table-demo__result">最近操作：{{ lastAction }}</div>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import { DasButton } from '@das-fed/upf-ui/packages/components/button'
import DasTable, { DasTableColumn } from '@das-fed/upf-ui/packages/components/table'

const data = [
  { id: 1, name: '基础应用', status: '运行中', owner: '平台组' },
  { id: 2, name: '组件文档', status: '建设中', owner: '前端组' },
  { id: 3, name: '微前端网关', status: '已发布', owner: '架构组' },
]
const loading = ref(false)
const highlightId = ref(2)
const selectionRows = ref<any[]>([])
const lastAction = ref('等待操作')
const operation = [
  { label: '查看', value: 'view' },
  { label: '发布', value: 'publish', disabled: (row: any) => row.status === '已发布' },
  { label: '更多', value: 'more', children: [{ label: '归档', value: 'archive' }] },
]

const handleOperation = (value: string, row: any) => {
  lastAction.value = `${value} · ${row.name}`
}
</script>

<style scoped>

  display: grid;

}

.table-demo__result {
  display: flex;
  flex-wrap: wrap;

  border: 1px solid var(--upf-border-color);
  border-radius: 6px;
  background: var(--upf-bg-secondary);
}
</style>
