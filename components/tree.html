Tree 树 ​

树组件。

导出 ​

DasTree、DasTreeEmpty、DasTreeRef

基础用法 ​

基础用法
展示静态树、多选、筛选、默认展开和操作项回调。
<template>
  <div class="tree-demo">
    <DasTree
      v-model="checkedKeys"
      v-model:filter-value="keyword"
      :data="treeData"
      node-key="id"
      :props="{ label: 'label', children: 'children' }"
      show-checkbox
      show-operations
      is-filter
      :default-expanded-keys="['platform']"
      height="260px"
      @change="handleChange"
      @handle-operation="handleOperation"
    />
    <div class="tree-demo__result">
      <span>关键词：{{ keyword || '全部' }}</span>
      <span>选中：{{ checkedKeys.join(' / ') || '无' }}</span>
      <span>{{ lastAction }}</span>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import { DasTree } from '@das-fed/upf-ui/packages/components/tree'

const treeData = [
  {
    id: 'platform',
    label: '平台中心',
    children: [
      { id: 'frontend', label: '前端组' },
      { id: 'framework', label: '框架组' },
    ],
  },
  {
    id: 'business',
    label: '业务中心',
    children: [
      { id: 'c3', label: 'C3 子应用' },
      { id: 'aoit', label: 'AOIT 子应用' },
    ],
  },
]
const checkedKeys = ref<string[]>(['frontend'])
const keyword = ref('')
const lastAction = ref('等待操作')

const handleChange = (value: string[]) => {
  checkedKeys.value = Array.isArray(value) ? value : [value]
}

const handleOperation = (item: { value: string }, _node: unknown, data: { label: string }) => {
  lastAction.value = `${item.value} · ${data.label}`
}
</script>

<style scoped>
.tree-demo {

}

}
</style>
