Form Item 表单项 ​

表单项组件，按 type 渲染输入、选择、日期、树选等控件。

导出 ​

DasFormItem

基础用法 ​

基础用法
展示 FormItem 的 label、冒号、必填、跨列和 v-model 聚合能力。
<template>
  <div class="form-item-basic-demo">
    <DasForm :cols="2" label-position="right" align-type="horizontal" content-width="220px">
      <DasFormItem v-model="form.name" label="名称" type="input" required show-colon />
      <DasFormItem v-model="form.status" label="状态" type="select" :options="options" show-colon />
      <DasFormItem v-model="form.enabled" label="启用" type="switch" show-colon />
      <DasFormItem v-model="form.remark" label="备注" type="textarea" :col="2" :autosize="{ minRows: 3 }" />
    </DasForm>
    <pre class="form-item-basic-demo__value">{{ form }}</pre>
  </div>
</template>

<script setup lang="ts">
import { reactive } from 'vue'
import { DasForm } from '@das-fed/upf-ui/packages/components/form'
import { DasFormItem } from '@das-fed/upf-ui/packages/components/form-item'

const form = reactive({ name: '', status: '', enabled: true, remark: '' })
const options = [
  { label: '启用', value: 'enabled' },
  { label: '停用', value: 'disabled' },
]
</script>

<style scoped>
.form-item-basic-demo {
  display: grid;
  gap: 14px;
}

.form-item-basic-demo__value {
  margin: 0;
  padding: 10px 12px;
  border-radius: 6px;
  color: var(--upf-text-secondary);
  background: var(--upf-bg-secondary);
  font-size: 12px;
  line-height: 20px;
}
</style>

帮助与校验 ​

帮助与校验
展示错误提示、帮助提示、tooltip 配置和输入框插槽。
<template>
  <div class="upf-demo-stack">
    <DasFormItem
      v-model="form.name"
      label="名称"
      type="input"
      required
      error="名称不能为空"
      help-message="名称会展示在页面标题和查询结果中"
      content-width="260px"
    />
    <DasFormItem
      v-model="form.status"
      label="状态"
      type="select"
      :options="options"
      :tooltip-attrs="{ content: '状态决定当前流程是否可继续使用' }"
      content-width="260px"
    />
    <DasFormItem
      v-model="form.owner"
      label="负责人"
      type="input"
      help-message="可通过插槽覆盖帮助内容"
      content-width="260px"
    >
      <template #helpContent>
        <div class="form-item-help-demo__tip">负责人会收到审批、催办和异常提醒。</div>
      </template>
      <template #prefix>
        <DasIcon icon="edit" icon-color="#909399" />
      </template>
    </DasFormItem>
  </div>
</template>

<script setup lang="ts">
import { reactive } from 'vue'
import { DasIcon } from '@das-fed/upf-ui/packages/components/icon'
import { DasFormItem } from '@das-fed/upf-ui/packages/components/form-item'

const form = reactive({
  name: '',
  status: '',
  owner: '',
})
const options = [
  { label: '启用', value: 'enabled' },
  { label: '停用', value: 'disabled' },
]
</script>

<style scoped>
.form-item-help-demo__tip {
  max-width: 180px;
  line-height: 20px;
}
</style>

不同类型 ​

不同类型
按 type 渲染输入、选择、单选、多选、开关、数字、颜色和自定义内容。
<template>
  <div class="form-item-type-demo">
    <DasForm :cols="2" label-position="right" align-type="horizontal" content-width="240px">
      <DasFormItem v-model="form.name" label="输入框" type="input" />
      <DasFormItem v-model="form.password" label="密码" type="password" />
      <DasFormItem v-model="form.status" label="选择器" type="select" :options="options" />
      <DasFormItem v-model="form.mode" label="单选" type="radio" :options="options" />
      <DasFormItem v-model="form.tags" label="多选" type="checkbox" :options="options" />
      <DasFormItem v-model="form.enabled" label="开关" type="switch" />
      <DasFormItem v-model="form.count" label="数字输入" type="inputNumber" :min="0" :max="99" />
      <DasFormItem v-model="form.color" label="颜色" type="colorPicker" />
      <DasFormItem v-model="form.remark" label="文本域" type="textarea" :col="2" :autosize="{ minRows: 3 }" />
      <DasFormItem label="自定义" type="custom" :col="2">
        <DasButton btn-type="primary" size="small" @click="form.name = '自定义填入'">写入名称</DasButton>
      </DasFormItem>
    </DasForm>
  </div>
</template>

<script setup lang="ts">
import { reactive } from 'vue'
import { DasButton } from '@das-fed/upf-ui/packages/components/button'
import DasForm from '@das-fed/upf-ui/packages/components/form'
import { DasFormItem } from '@das-fed/upf-ui/packages/components/form-item'

const form = reactive({
  name: '',
  password: '',
  status: '',
  mode: 'enabled',
  tags: ['enabled'],
  enabled: true,
  count: 3,
  color: '#315efb',
  remark: '',
})
const options = [
  { label: '启用', value: 'enabled' },
  { label: '停用', value: 'disabled' },
]
</script>

<style scoped>
.form-item-type-demo {
  width: 100%;
}
</style>
