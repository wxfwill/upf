Edit Table 可编辑表格 ​

可编辑表格组件，适合表格内录入和校验。

导出 ​

DasEditTable、DasEditTableRef

基础用法 ​

基础用法
使用 config 配置可编辑列，展示输入、选择、数字、插入、删除和整表校验。
<template>
  <div class="edit-table-demo">
    <div class="upf-demo-row">
      <DasButton @click="addRow">新增行</DasButton>
      <DasButton btn-type="default" @click="validateAll">校验全部</DasButton>
      <DasButton btn-type="default" @click="removeFirst">删除首行</DasButton>
    </div>
    <DasEditTable ref="tableRef" :data="data" :config="config" row-key="id" @add-row="lastAction = '触发新增事件'" />
    <pre class="edit-table-demo__result">{{ lastAction }}</pre>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import DasButton from '@das-fed/upf-ui/packages/components/button'
import { DasEditTable } from '@das-fed/upf-ui/packages/components/edit-table'

const tableRef = ref<any>()
const lastAction = ref('点击单元格可进入编辑')
const statusOptions = [
  { id: 'enabled', name: '启用' },
  { id: 'disabled', name: '停用' },
]
const config = [
  {
    prop: 'name',
    label: '名称',
    mode: 'input',
    required: true,
    rules: [{ required: true, message: '请输入名称' }],
    placeholder: '请输入名称',
    change: (_val: any, row: any) => {
      lastAction.value = `名称变更：${row.name}`
    },
  },
  {
    prop: 'status',
    label: '状态',
    mode: 'select',
    options: statusOptions,
    optionsValueKey: 'id',
    optionsLabelKey: 'name',
    rules: [{ required: true, message: '请选择状态' }],
  },
  {
    prop: 'count',
    label: '数量',
    mode: 'inputNumber',
    inputWidth: '120px',
    rules: [{ required: true, message: '请输入数量' }],
  },
]
const data = ref([
  { id: 1, name: '任务 A', status: 'enabled', count: 2 },
  { id: 2, name: '任务 B', status: 'disabled', count: 1 },
])

const addRow = () => {
  tableRef.value?.insert?.({ id: Date.now(), name: '', status: 'enabled', count: 1 })
  lastAction.value = '已插入一行'
}

const removeFirst = () => {
  if (!data.value.length) return
  tableRef.value?.removeRowByKey?.(data.value[0].id)
  lastAction.value = '已删除首行'
}

const validateAll = () => {
  tableRef.value?.validateAll?.((valid: boolean) => {
    lastAction.value = valid ? '校验通过' : '校验未通过'
  })
}
</script>

<style scoped>
.edit-table-demo {
  display: grid;
  gap: 12px;
}

.edit-table-demo__result {
  margin: 0;
  color: var(--upf-text-secondary);
  font-size: 13px;
  white-space: pre-wrap;
}
</style>
