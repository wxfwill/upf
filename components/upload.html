DasUpload

基础用法 ​

基础用法
展示文件上传、错误提示与文件列表清空等常见导入场景。
<template>
  <div class="upload-demo">
    <DasUpload
      v-model="fileList"
      :max-count="3"
      accept=".xlsx,.xls,.csv"
      error-message="部分数据校验不通过，请修正后重新导入。"
      :error="error"
      :error-file-name="errorFileName"
      :error-file-size="errorFileSize"
      @file-upload="handleUpload"
    />
    <div class="upload-demo__toolbar">
      <DasButton size="small" @click="reset">清空文件列表</DasButton>
      <span>{{ tip }}</span>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import type { UploadProps } from 'ant-design-vue'
import { DasButton, DasUpload } from '@das-fed/upf-ui'

const fileList = ref<UploadProps['fileList']>([])
const error = ref(false)
const errorFileName = ref('')
const errorFileSize = ref('')
const tip = ref('演示多文件上传、错误提示与清空操作。')

const handleUpload = (file: { name?: string; size?: number }) => {
  error.value = true
  errorFileName.value = file?.name ?? '导入文件.xlsx'
  errorFileSize.value = file?.size ? `${Math.ceil(file.size / 1024)}KB` : '128KB'
  tip.value = '已模拟一次上传校验失败'
}

const reset = () => {
  fileList.value = []
  error.value = false
  errorFileName.value = ''
  errorFileSize.value = ''
  tip.value = '文件列表已清空'
}
</script>

<style scoped>
.upload-demo {
  display: grid;
  gap: 12px;
}

.upload-demo__toolbar {
  display: flex;
  flex-wrap: wrap;
  align-items: center;
  gap: 8px 12px;
  color: var(--upf-text-secondary);
  font-size: 13px;
}
</style>
