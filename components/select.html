Select 选择器 ​

选择器组件，基于 Ant Design Vue Select 增强只读、错误态、全选和缺失 label 回显。

导出 ​

DasSelect、DasSelectOption

基础用法 ​

基础用法
展示单选、多选、字段映射和 change 回显。
<template>
  <div class="select-demo">
    <DasSelect v-model="value" :options="options" placeholder="请选择状态" width="260px" @change="handleChange" />
    <DasSelect
      v-model="mappedValue"
      :options="mappedOptions"
      :field-names="{ label: 'name', value: 'id' }"
      placeholder="字段映射"
      width="260px"
    />
    <DasSelect v-model="multiValue" mode="multiple" :options="options" placeholder="请选择多个状态" width="320px" />
    <div class="select-demo__result">
      <span>单选：{{ value || '未选择' }}</span>
      <span>字段映射：{{ mappedValue }}</span>
      <span>多选：{{ multiValue.join(' / ') || '未选择' }}</span>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import { DasSelect } from '@das-fed/upf-ui/packages/components/select'

const value = ref()
const mappedValue = ref('admin')
const multiValue = ref<string[]>([])
const options = [
  { label: '启用', value: 'enabled' },
  { label: '停用', value: 'disabled' },
  { label: '草稿', value: 'draft' },
]
const mappedOptions = [
  { name: '管理员', id: 'admin' },
  { name: '运营', id: 'ops' },
  { name: '访客', id: 'guest' },
]

const handleChange = (nextValue: string) => {
  value.value = nextValue
}
</script>

状态展示 ​

状态展示
展示默认、查看、禁用、错误和缺失选项 label 保留。
<template>
  <div class="select-state-demo">
    <DasSelect v-model="value" :options="options" placeholder="默认状态" width="260px" allow-clear />
    <DasSelect v-model="viewValue" :options="options" is-view width="260px" />
    <DasSelect v-model="disabledValue" :options="options" disabled width="260px" />
    <DasSelect v-model="errorValue" :options="options" error error-label="请选择状态" width="260px" />
    <DasSelect
      v-model="missingValue"
      :options="missingOptions"
      preserve-label-when-missing
      width="260px"
      placeholder="缺失 label 回显"
    />
    <DasButton btn-type="default" size="small" @click="toggleMissingOptions">
      {{ missingOptions.length === optionsWithLegacy.length ? '模拟缺失选项' : '恢复选项' }}
    </DasButton>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import { DasButton } from '@das-fed/upf-ui/packages/components/button'
import { DasSelect } from '@das-fed/upf-ui/packages/components/select'

const value = ref()
const viewValue = ref('enabled')
const disabledValue = ref('draft')
const errorValue = ref()
const missingValue = ref('legacy')
const options = [
  { label: '启用', value: 'enabled' },
  { label: '停用', value: 'disabled' },
  { label: '草稿', value: 'draft' },
]
const optionsWithLegacy = [...options, { label: '历史选项', value: 'legacy' }]
const missingOptions = ref(optionsWithLegacy)

const toggleMissingOptions = () => {
  missingOptions.value = missingOptions.value.length === optionsWithLegacy.length ? options : optionsWithLegacy
}
</script>

<style scoped>
.select-state-demo {
  display: grid;
  gap: 14px;
  align-items: start;
}
</style>

多选与标签 ​

多选与标签
展示多选全选、tags 输入、大批量选择折叠和外部控制标签展示。
<template>
  <div class="select-tags-demo">
    <DasSelect
      ref="multipleSelectRef"
      v-model="multipleValue"
      mode="multiple"
      :options="options"
      width="320px"
      placeholder="多选"
      :show-select-all="true"
      :is-exter-control-count="true"
      @focus="handleSelectFocus"
      @blur="handleSelectBlur"
    />

    <DasSelect
      ref="tagSelectRef"
      v-model="tagValue"
      mode="tags"
      :options="options"
      width="320px"
      placeholder="标签模式"
      :max-tag-text-length="4"
      :is-exter-control-count="true"
      @focus="handleSelectFocus"
      @blur="handleSelectBlur"
    />
    <DasSelect
      v-model="largeValue"
      mode="multiple"
      :options="largeOptions"
      width="320px"
      placeholder="大批量选择折叠"
      :large-selection-collapse-threshold="5"
    />
    <div class="select-tags-demo__toolbar">
      <DasButton btn-type="default" size="small" @click="multipleValue = options.map((item) => item.value)">
        全选状态
      </DasButton>
      <DasButton btn-type="default" size="small" @click="multipleValue = []">清空</DasButton>
      <span>已选 {{ multipleValue.length }} 项</span>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import { DasButton } from '@das-fed/upf-ui/packages/components/button'
import { DasSelect } from '@das-fed/upf-ui/packages/components/select'

const multipleValue = ref(['enabled'])
const tagValue = ref(['前端平台', '组件文档'])
const largeValue = ref(['item-1', 'item-2', 'item-3', 'item-4', 'item-5', 'item-6'])
const multipleSelectRef = ref<any>(null)
const tagSelectRef = ref<any>(null)
const options = [
  { label: '启用', value: 'enabled' },
  { label: '停用', value: 'disabled' },
  { label: '草稿', value: 'draft' },
]
const largeOptions = Array.from({ length: 12 }, (_, index) => ({
  label: `选项 ${index + 1}`,
  value: `item-${index + 1}`,
}))

const syncDisplayMode = (isFocus: boolean) => {
  multipleSelectRef.value?.toChangeMaxCount?.(isFocus)
  tagSelectRef.value?.toChangeMaxCount?.(isFocus)
}

const handleSelectFocus = () => {
  syncDisplayMode(true)
}

const handleSelectBlur = () => {
  // 文档示例期望持续展示已选标签，避免失焦后折叠成计数文案。
  syncDisplayMode(true)
}
</script>

<style scoped>
.select-tags-demo {
  display: grid;
  gap: 14px;
  align-items: start;
}

.select-tags-demo__toolbar {
  display: flex;
  flex-wrap: wrap;
  align-items: center;
  gap: 10px;
  color: var(--upf-text-secondary);
  font-size: 13px;
}
</style>
