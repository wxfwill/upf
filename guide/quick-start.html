快速开始 ​

全局注册 ​

ts
import { createApp } from 'vue'
import App from './App.vue'
import DasUI from '@das-fed/upf-ui'
import '@das-fed/upf-web/packages/style'

const app = createApp(App)

app.use(DasUI)
app.mount('#app')

<template>

import { DasButton } from '@das-fed/upf-ui'

表单与表格组合 ​

vue
<template>
  <DasForm :cols="2">
    <DasFormItem label="名称" type="input" v-model="form.name" />
    <DasFormItem label="状态" type="select" v-model="form.status" :options="statusOptions" />
  </DasForm>

  <DasTable :data="tableData" :total="100">
    <DasTableColumn prop="name" label="名称" />
    <DasTableColumn prop="status" label="状态" />
  </DasTable>
</template>

<script setup lang="ts">
import { reactive } from 'vue'
import { DasForm, DasFormItem, DasTable, DasTableColumn } from '@das-fed/upf-ui'

const form = reactive({ name: '', status: '' })
const statusOptions = [{ label: '启用', value: 'enabled' }]
const tableData = [{ name: '项目 A', status: '启用' }]
</script>
